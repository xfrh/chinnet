ALTER PROC [dbo].[GetMedicalAntibioticResult_Province]
 @OrganismId BIGINT , --　本次查询细菌的Id
  @ProvinceId BIGINT --　本次查询省份的Id
AS

DECLARE @OrganismName NVARCHAR(1000) -- 本次查询细菌的名称
DECLARE @OrganismCode NVARCHAR(1000)　-- 本次查询的细菌编码

-- 查询的细菌
SELECT @OrganismId= Id, @OrganismCode=Code,@OrganismName=Name FROM  dbo.MedicalOrganism WHERE Id = @OrganismId 

--为查询到细菌数据，直接返回
IF @OrganismId <= 0
BEGIN
	RETURN
END

 --声明临时表，记录本次需要处理的数据
 SELECT  * INTO  #DataTable  FROM  MedicalAntibioticResult  WHERE Mark>0 AND OrganismId=@OrganismId  AND IsValid=1 
 AND MedicalDataId IN ( SELECT Id FROM  dbo.MedicalData WHERE Status =1 AND AreaId = @ProvinceId )


 --声明临时表存数据
 CREATE TABLE #ResultTable
 (
    FieldName NVARCHAR(1000), --字段名称，也就是抗生素对应的导入列名称
	OrganismId BIGINT , -- 细菌Id
	OrganismName NVARCHAR(1000), --细菌名称
	OrganismCode NVARCHAR(1000), --细菌编码
	AntibioticId BIGINT , -- 抗生素Id
	AntibioticName NVARCHAR(1000), --抗生素名称
	AntibioticCode NVARCHAR(1000), --抗生素编码
	SensitiveCount INT, -- 敏感数量
	IntermediaryCount INT, -- 中介数量
	ResistanceCount INT, -- 耐药数量
 )
